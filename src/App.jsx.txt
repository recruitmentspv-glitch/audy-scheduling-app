const getWeekDates = (weekOffset = 0) => {
  const dates = [];
  const today = new Date();
  const dayOfWeek = today.getDay(); 
  const diffToMonday = today.getDate() - dayOfWeek + (dayOfWeek === 0 ? -6 : 1);
  
  const monday = new Date(today);
  monday.setDate(diffToMonday + (weekOffset * 7));
  
  for (let i = 0; i < 5; i++) { // Senin - Jumat
    const d = new Date(monday);
    d.setDate(monday.getDate() + i);
    
    const days = ['Minggu', 'Senin', 'Selasa', 'Rabu', 'Kamis', 'Jumat', 'Sabtu'];
    const months = ['Jan', 'Feb', 'Mar', 'Apr', 'Mei', 'Jun', 'Jul', 'Ags', 'Sep', 'Okt', 'Nov', 'Des'];
    
    const dateStr = d.toISOString().split('T')[0];
    const label = `${days[d.getDay()]}, ${d.getDate()} ${months[d.getMonth()]}`;
    
    dates.push({ id: dateStr, label, weekOffset });
  }
  return dates;
};

const ALL_DATES = [

// --- DATA GENERATOR ---
const generateSlots = () => {
  const startTimes = [
    '09:00', '09:20', '09:40', '10:00', '10:20', '10:40',
    '11:00', '11:20', '11:40', '13:00', '13:20', '13:40',
    '14:00', '14:20', '14:40'
  ];
  
  let allSlots = [];
  ALL_DATES.forEach(dateObj => {
    startTimes.forEach((time, i) => {
      const [h, m] = time.split(':').map(Number);
      const d = new Date(); 
      d.setHours(h, m + 20);
      const endTime = `${String(d.getHours()).padStart(2,'0')}:${String(d.getMinutes()).padStart(2,'0')}`;
      
      allSlots.push({
        id: `${dateObj.id}-${i}`,
        dateId: dateObj.id,
        dateLabel: dateObj.label,
        time: `${time} - ${endTime}`,
        recruitersState: {} 
      });
    });
  });
  return allSlots;
};

// Utils: Mengubah file menjadi format teks panjang (Base64) agar bisa dikirim via internet
const fileToBase64 = (file) => new Promise((resolve, reject) => {
  const reader = new FileReader();
  reader.readAsDataURL(file);
  reader.onload = () => resolve(reader.result);
  reader.onerror = error => reject(error);
});

// --- CANDIDATE VIEW ---
function CandidateView({ slots, onBookSlot }) {

  const [selectedSlot, setSelectedSlot] = useState(null);
  const [candidateForm, setCandidateForm] = useState({ name: '', email: '', file: null });
  const [successData, setSuccessData] = useState(null);
  const [isSubmitting, setIsSubmitting] = useState(false);

  const daySlots = slots.filter(s => s.dateId === selectedDateId);

  const handleBook = async (e) => {
    e.preventDefault();
    if (candidateForm.name && candidateForm.email && selectedSlot) {
      setIsSubmitting(true);
      
      try {
        let fileBase64 = "";
        let fileName = "";
        
        // Membaca file jika ada
        if (candidateForm.file) {
          fileBase64 = await fileToBase64(candidateForm.file);
          fileName = candidateForm.file.name;
        }

        // 1. Eksekusi booking secara lokal untuk mendapatkan Meet Link & Nama Rekruter yang bertugas
        const { assignedRecruiter, meetLink } = onBookSlot(selectedSlot.id, { 
          name: candidateForm.name, 
          email: candidateForm.email,
          fileName: fileName
        });
        
        // 2. Siapkan data untuk dikirim ke Google Sheets (Backend)
        const payload = {
          candidateName: candidateForm.name,
          candidateEmail: candidateForm.email,
          scheduleTime: `${selectedSlot.dateLabel}, Pukul ${selectedSlot.time}`,
          fileBase64: fileBase64,
          fileName: fileName,
          meetLink: meetLink,
          recruiterName: assignedRecruiter?.name || "Tidak ada"
        };

        // 3. POST data ke Google Apps Script URL
        await fetch(GOOGLE_SCRIPT_URL, {
          method: "POST",
          body: JSON.stringify(payload)
        });

        // 4. Tampilkan layar sukses
        setSuccessData({
          ...selectedSlot,
          candidateName: candidateForm.name,
          assignedRecruiter,
          meetLink
        });
        
      } catch (error) {
        console.error("Terjadi kesalahan:", error);
        alert("Mohon maaf, terjadi kendala saat mengirim data ke server. Pastikan koneksi stabil.");
      } finally {
        setIsSubmitting(false);
        setSelectedSlot(null);
        setCandidateForm({ name: '', email: '', file: null });
      }
    }
  };

  if (successData) {
    return (
      <div className="max-w-2xl mx-auto animate-in zoom-in-95 duration-300">
        <div className="bg-white rounded-2xl shadow-lg border border-slate-200 overflow-hidden">
          <div className="bg-emerald-500 p-8 text-center text-white">
            <div className="w-16 h-16 bg-white/20 rounded-full flex items-center justify-center mx-auto mb-4 backdrop-blur-sm">
              <CheckCircle className="w-10 h-10" />
            </div>
            <h2 className="text-2xl font-bold mb-2">Jadwal Berhasil Dikonfirmasi!</h2>
            <p className="text-emerald-50">Terima kasih {successData.candidateName}, jadwal interview Anda telah tersimpan.</p>
          </div>
          
          <div className="p-8 space-y-6">
            <div className="grid grid-cols-2 gap-4">
              <div className="p-4 bg-slate-50 rounded-xl border border-slate-100">
                <div className="text-sm font-semibold text-slate-500 uppercase tracking-wider mb-1 flex items-center gap-2"><Calendar className="w-4 h-4"/> Tanggal</div>
                <div className="text-lg font-bold text-slate-800">{successData.dateLabel}</div>
              </div>
              <div className="p-4 bg-slate-50 rounded-xl border border-slate-100">
                <div className="text-sm font-semibold text-slate-500 uppercase tracking-wider mb-1 flex items-center gap-2"><Clock className="w-4 h-4"/> Pukul</div>
                <div className="text-lg font-bold text-slate-800">{successData.time} WIB</div>
              </div>
            </div>

            <div className="p-5 border-2 border-indigo-100 bg-indigo-50/50 rounded-xl">
              <div className="flex items-center gap-2 font-bold text-indigo-800 mb-3 text-lg">
                <Video className="w-5 h-5" /> Link Google Meet
              </div>
              <a href={successData.meetLink} target="_blank" rel="noopener noreferrer" className="block w-full p-3 bg-white border border-indigo-200 rounded-lg text-indigo-600 font-medium text-center hover:bg-indigo-50 transition-colors break-all mb-4">
                {successData.meetLink}
              </a>
              
              <div className="flex gap-3 items-start bg-amber-50 p-4 rounded-lg border border-amber-200 text-amber-900">
                <ShieldAlert className="w-5 h-5 shrink-0 text-amber-600 mt-0.5" />
                <div className="text-sm leading-relaxed">
                  <strong>Mohon masuk ke link Google Meet tepat waktu.</strong> Anda akan masuk ke ruang tunggu (waiting room) terlebih dahulu dan menunggu persetujuan host untuk masuk, karena mungkin interview sebelumnya belum selesai.
                </div>
              </div>
            </div>

            <div className="flex items-start gap-3 p-4 bg-slate-50 rounded-xl text-slate-600 border border-slate-100">
              <Bell className="w-5 h-5 shrink-0 text-indigo-400 mt-0.5" />
              <div className="text-sm">
                <strong>Reminder otomatis</strong> akan dikirimkan pada pukul <strong>08:00 pagi</strong> di hari H interview.
              </div>
            </div>

            <button 
              onClick={() => setSuccessData(null)}
              className="w-full py-3 bg-slate-900 hover:bg-slate-800 text-white rounded-xl font-medium transition-colors mt-4"
            >
              Selesai & Kembali
            </button>
          </div>
        </div>
      </div>
    );
  }

  return (
    <div className="space-y-6 animate-in fade-in duration-300">
      <div className="text-center max-w-2xl mx-auto mb-8">
        <h1 className="text-3xl font-bold text-slate-900 mb-3">Pilih Jadwal Wawancara</h1>
        <p className="text-slate-600">
          Pilih tanggal dan slot waktu yang tersedia untuk sesi wawancara Anda bersama tim <strong>Audy Dental</strong>.
        </p>
      </div>

      <div className="flex justify-start sm:justify-center overflow-x-auto gap-3 mb-8 pb-2 scrollbar-hide px-2">
        {visibleDates.map(dateObj => (
          <button
            key={dateObj.id}
            onClick={() => {
              setSelectedDateId(dateObj.id);
              setSelectedSlot(null);
            }}
            className={`px-5 py-3 rounded-2xl text-sm font-semibold transition-all border shadow-sm flex-shrink-0 whitespace-nowrap ${
              selectedDateId === dateObj.id 
                ? 'border-indigo-600 bg-indigo-600 text-white shadow-md transform scale-105' 
                : 'border-slate-200 bg-white text-slate-600 hover:border-indigo-300 hover:text-indigo-600'
            }`}
          >
            {dateObj.label}
          </button>
        ))}
      </div>

      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-5">
        {daySlots.map(slot => {
          const availableCount = Object.values(slot.recruitersState).filter(state => !state.isBooked).length;
          const totalOpen = Object.values(slot.recruitersState).length;
          const isDisabled = availableCount === 0;
          const isSelected = selectedSlot?.id === slot.id;

          return (
            <div 
              key={slot.id} 
              className={`bg-white border rounded-xl overflow-hidden transition-all duration-200 ${
                isSelected 
                  ? 'border-indigo-500 ring-4 ring-indigo-500/10 shadow-md' 
                  : isDisabled
                    ? 'border-slate-100 bg-slate-50 opacity-60'
                    : 'border-slate-200 shadow-sm hover:border-indigo-300 hover:shadow-md cursor-pointer'
              }`}
              onClick={() => {
                if (!isDisabled && !isSelected) setSelectedSlot(slot);
              }}
            >
              <div className="p-5">
                <div className="flex justify-between items-center mb-4">
                  <div className={`flex items-center font-bold text-lg ${isDisabled ? 'text-slate-500' : 'text-slate-900'}`}>
                    <Clock className={`w-5 h-5 mr-2 ${isDisabled ? 'text-slate-400' : 'text-indigo-500'}`} />
                    {slot.time}
                  </div>
                  
                  {availableCount > 0 ? (
                    <span className="text-xs font-semibold px-2 py-1 bg-emerald-50 border border-emerald-200 text-emerald-700 rounded-md">
                      Tersedia {availableCount} Slot
                    </span>
                  ) : totalOpen > 0 ? (
                    <span className="text-xs font-semibold px-2 py-1 bg-slate-200 text-slate-600 rounded-md">
                      Penuh
                    </span>
                  ) : (
                    <span className="text-xs font-semibold px-2 py-1 bg-slate-100 text-slate-400 rounded-md">
                      Tutup
                    </span>
                  )}
                </div>

                {isSelected ? (
                  <div className="animate-in fade-in zoom-in-95 duration-200 pt-2 border-t border-slate-100 mt-2">
                    <form onSubmit={handleBook} className="space-y-3 mt-3">
                      <div>
                        <input 
                          type="text" 
                          required
                          disabled={isSubmitting}
                          placeholder="Nama Lengkap Kandidat"
                          value={candidateForm.name}
                          onChange={(e) => setCandidateForm({...candidateForm, name: e.target.value})}
                          className="w-full text-sm rounded-md border border-slate-300 px-3 py-2.5 focus:outline-none focus:ring-2 focus:ring-indigo-500 disabled:opacity-50"
                          autoFocus
                        />
                      </div>
                      <div>
                        <input 
                          type="email" 
                          required
                          disabled={isSubmitting}
                          placeholder="Alamat Email Valid"
                          value={candidateForm.email}
                          onChange={(e) => setCandidateForm({...candidateForm, email: e.target.value})}
                          className="w-full text-sm rounded-md border border-slate-300 px-3 py-2.5 focus:outline-none focus:ring-2 focus:ring-indigo-500 disabled:opacity-50"
                        />
                      </div>
                      <div>
                        <label className="block text-xs font-medium text-slate-500 mb-1 mt-1">Upload CV / Form Lamaran (PDF)</label>
                        <input 
                          type="file" 
                          accept=".pdf,.doc,.docx"
                          required
                          disabled={isSubmitting}
                          onChange={(e) => setCandidateForm({...candidateForm, file: e.target.files[0]})}
                          className="w-full text-sm rounded-md border border-slate-300 px-3 py-2 focus:outline-none focus:ring-2 focus:ring-indigo-500 file:mr-4 file:py-1 file:px-3 file:rounded-md file:border-0 file:text-xs file:font-semibold file:bg-indigo-50 file:text-indigo-700 hover:file:bg-indigo-100 disabled:opacity-50"
                        />
                      </div>
                      <div className="flex gap-2 pt-3">
                        <button 
                          type="button"
                          disabled={isSubmitting}
                          onClick={(e) => {
                            e.stopPropagation();
                            setSelectedSlot(null);
                          }}
                          className="flex-1 px-3 py-2.5 text-sm font-medium text-slate-600 bg-slate-100 rounded-lg hover:bg-slate-200 transition-colors disabled:opacity-50"
                        >
                          Batal
                        </button>
                        <button 
                          type="submit"
                          disabled={isSubmitting}
                          className="flex-1 px-3 py-2.5 text-sm font-medium text-white bg-indigo-600 rounded-lg hover:bg-indigo-700 transition-colors shadow-sm flex justify-center items-center gap-2 disabled:opacity-75 disabled:cursor-not-allowed"
                        >
                          {isSubmitting ? (
                            <><Loader2 className="w-4 h-4 animate-spin" /> Memproses...</>
                          ) : (
                            'Konfirmasi'
                          )}
                        </button>
                      </div>
                    </form>
                  </div>
                ) : (
                  <button 
                    disabled={isDisabled}
                    className={`w-full mt-2 py-2.5 px-4 text-sm font-semibold rounded-lg transition-colors border flex justify-center items-center gap-1 group ${
                      isDisabled 
                        ? 'bg-slate-100 text-slate-400 border-transparent cursor-not-allowed'
                        : 'bg-indigo-50 hover:bg-indigo-100 text-indigo-700 border-indigo-100'
                    }`}
                  >
                    {isDisabled ? 'Tidak Tersedia' : 'Pilih Jadwal'}
                    {!isDisabled && <ChevronRight className="w-4 h-4 group-hover:translate-x-1 transition-transform" />}
                  </button>
                )}
              </div>
            </div>
          );
        })}
      </div>
    </div>
  );
}
